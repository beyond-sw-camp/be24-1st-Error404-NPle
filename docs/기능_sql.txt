    p.id, 
    p.name AS product_name, 
    p.price, 
    p.stock, 
    c.name AS category_name, 
    s.name AS store_name
FROM product p
JOIN category c ON p.category_id = c.id
JOIN store s ON p.store_id = s.id

## 상품 상세 조회_USER_006
SELECT 
    p.id AS product_id,
    p.name AS product_name,
    p.price,
    c.name AS category_name,
    s.name AS store_name,
    r.contents AS review_content,
    r.rating AS review_rating,
    u.name AS reviewer_name
FROM product p
JOIN category c ON p.category_id = c.id
JOIN store s ON p.store_id = s.id
LEFT JOIN review r ON p.id = r.product_id
LEFT JOIN user u ON r.user_id = u.id
WHERE p.id = 30; -- 조회할 상품 ID

SELECT product.name AS PRODUCT, AVG(review.rating) AS AVG_RATING FROM store
INNER JOIN product ON store.id = product.store_id
INNER JOIN review ON review.product_id = product.id

## 상품 상세 페이지 진입 시 공동구매 상태 및 게이지 데이터 조회_DEAL_001
SELECT 
    d.id AS deal_id,
    p.name AS product_name,
    p.price AS original_price,
    d.discountRate,
    ## 실시간 할인가 계산
    ROUND(p.price * (1 - (d.discountRate / 100))) AS current_deal_price,
    d.participants AS goal_count,
    d.current_participants_count AS current_count,
    ## 달성률(%) 계산 (DEAL_001 로직)
    ROUND((d.current_participants_count / d.participants) * 100, 1) AS achievement_rate,
    d.finished_at,
    d.status
FROM deal d
JOIN product p ON d.product_id = p.id
WHERE d.id = [조회할_DEAL_ID];

## 공유하기 썸네일 생성용 데이터 조회_DEAL_005
SELECT 
    p.name,
    d.discountRate,
    (d.participants - d.current_participants_count) AS remaining_count,
    CONCAT('현재 ', d.discountRate, '% 할인 중! ', 
           (d.participants - d.current_participants_count), '명 남았습니다.') AS share_text
FROM deal d
JOIN product p ON d.product_id = p.id
WHERE d.id = [공유할_딜_ID];
